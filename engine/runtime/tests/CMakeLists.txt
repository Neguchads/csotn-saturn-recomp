# The runtime's unit tests, built on their own: no recompiled game, no SDL needed.
#   cmake -S runtime/tests -B build/runtime-tests -G Ninja -DCMAKE_CXX_COMPILER=clang++
#   ninja -C build/runtime-tests && build/runtime-tests/runtime_tests
cmake_minimum_required(VERSION 3.20)
project(saturn_runtime_tests CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../runtime.cmake)
add_executable(runtime_tests main.cpp no_modules.cpp vdp2_test.cpp bup_test.cpp pads_test.cpp settings_test.cpp touch_test.cpp pace_test.cpp frame_change_test.cpp modem_test.cpp scu_test.cpp cdrom_test.cpp cdblock_test.cpp clock_test.cpp onchip_test.cpp tasks_test.cpp scsp_test.cpp vdp1_timing_test.cpp video_capture_test.cpp)
target_link_libraries(runtime_tests saturn_core saturn_machine saturn_host Threads::Threads)
add_executable(video_perf_tests video_perf_test.cpp)
target_include_directories(video_perf_tests PRIVATE ../src/video)
target_link_libraries(video_perf_tests Threads::Threads)
if(MSVC)
  target_compile_options(video_perf_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(video_perf_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME runtime_tests COMMAND runtime_tests)
add_test(NAME video_perf_tests COMMAND video_perf_tests)

# GPU real opcional: sem janela, ROM, jogo ou save. Exige a SDL já instalada.
option(SATURN_VDP2_GPU_TESTS "comparar VDP2 GPU/CPU com SDL3" OFF)
if(SATURN_VDP2_GPU_TESTS)
  find_package(SDL3 3.4.14 CONFIG REQUIRED)
  add_executable(vdp2_gpu_tests vdp2_gpu_test.cpp vdp2_gpu_compare.cpp vdp2_test.cpp)
  target_include_directories(vdp2_gpu_tests PRIVATE . ../include ../src/video ../src/host ../src/machine)
  target_link_libraries(vdp2_gpu_tests SDL3::SDL3)
  add_test(NAME vdp2_gpu_direct3d12 COMMAND vdp2_gpu_tests direct3d12)
  add_test(NAME vdp2_gpu_vulkan COMMAND vdp2_gpu_tests vulkan)
endif()
if(SATURN_VDP1_GPL)
  target_compile_definitions(runtime_tests PRIVATE SATURN_VDP1_GPL=1)
endif()
